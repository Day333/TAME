#  univariate:   data/Uni/<source_name>/*.csv 或 *.out
#  multivariate:   data/Multi/<source_name>/*.csv 或 *.out 或 *.rds

#  支持的原始格式:
#     (a) 两列 "value,label"         —— 有无表头都可以, 脚本自动识别
#     (b) 宽表 "ch1,ch2,...,Label"   —— 每列一个通道, 最后一列是标签
#     (c) 多文件 "@N" 通道         —— 如 "100@1.out" / "100@2.out", 脚本按基础名自动拼成一个多通道序列
#     (d) R 数据文件 .rds
